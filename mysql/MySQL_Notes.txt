//UNIQUE Constraint is something that prevent duplicate entries in a specified column, Consider the following example;
    CREATE TABLE products(
                        product_id INT,

    INSERT INTO products VALUES (100, "hamburger", 3.99),

    ALTER TABLE products ADD CONSTRAINT UNIQUE(product_name);       //Add the UNIQUE Constraint to the product_name column, after column creation

//CHECK Constraint is used to check if certain criteria a met when adding table entries
    CREATE TABLE employees (
        employee_id INT,
        first_name VARCHAR(50),
        last_name VARCHAR(50),
        hourly_pay DECIMAL(5, 2),
        hire_date DATE,
        CONSTRAINT chk_hourly_pay CHECK (hourly_pay >= 10.00)       //Suppose the minimum employees wage is 10.00, so if some entry is added which is below 10.00 an error will be generated
    );
    //The CHECK constraint can be given a name for easy reference in the future if any changes need to be done, for this we use the CONSTRAINT and the name of the CHECK as "chk_hourly_pay"

    ALTER TABLE employees ADD CONSTARAINT chk_hourly_pay CHECK(hourly_pay >= 10.00);    //Add the CHECK constraint to the hourly_pay column, after column creation
    ALTER TABLE employees DROP CHECK chk_hourly_pay;                                    //Remove the CHECK constraint from the employees table

//Foreign Keys - A PRIMARY KEY from one table that can be found within a different table. Using a foreign key we can establish connection between two tables
    When the a foreign key constraint is made that would create a link between two tables, which would prevent any actions that would destroy the link between them
    So for foreign keys to work there should be two or more tables, that would have similar columns

    eg: Suppose a there are two tables;
    Table1 transactions -   transaction_id (PRIMARY KEY)
                            amount
                            customer_id (Foreign Key)

    Table2 customers    -   customer_id (PRIMARY KEY)
                            first_name
                            last_name

    CREATE TABLE customers(
                customer_id INT PRIMARY KEY AUTO_INCREMENT,
                first_name VARCHAR(50),
                last_name VARCHAR(50),
    );

    CREATE TABLE transactions (
                transaction_id INT PRIMARY KEY AUTO_INCREMENT,
                amount DECIMAL(5, 2),
                customer_id INT,
                FOREIGN KEY (customer_id) REFERENCES customers(customer_id)
    );
    //This will create a foreign key constaraint for the customer_id column of the transactions and link to the customer_id of the customers table
    //Any foreign keys of a table can be found within the side panel under foreign keys tab of a table, if not speacified the server will assign a name for created foreign key. This name is useful in making further changes after the foreign key is created

    ALTER TABLE transactions DROP FOREIGN KEY transactions_ibfk_1;                                                          //This will drop the foreign key
    ALTER TABLE transactions FOREIGN KEY (customer_id) REFERENCES customers(customer_id)                                    //Created foreign keys with the default anme given by the server
    ALTER TABLE transactions ADD CONSTRAINT fk_customer_id FOREIGN KEY (customer_id) REFERENCES customers(customer_id)      //Created foreign keys with the name fk_customer_id

So now suppose we populate the customers table, and if we then try to do something like DELETE FROM customers WHERE customer_id=3; we are not allowed to do this unless we drop the foreign key

//Joins in MySQL
A join is a clause that is used to combine rows of two or more tables based on a related column between them such as a foreign key
We can think of it as a two sets in a venn diagram that has a common area. So we assign tables to left side set and another table to right side set and there is common data field where the two sets intersect, these are called left joins, right joins and inner joins respectively. So ultimately joins is a method used as a data parsing method for ease of use

    SELECT * FROM transactions INNER JOIN customers ON transactions.customer_id = customers.customer_id;
    //This will give an output that has transactions table with all its content on the left and customers table with all its content on the right. So basically we have an output that has both tables joint with all the fields from each table. So entry that does not share data with both that tables will be excluded from the output(Inner join)(transactions n customers)

    SELECT transaction_id, amount, first_name, last_name FROM transactions INNER JOIN customers ON transactions.customer_id = customers.customer_id;
    //This shows the only the transaction_id, amount, first_name, last_name from the above joint table (Inner join)

    SELECT * FROM transactions LEFT JOIN customers ON transactions.customer_id = customers.customer_id;
    //This will pull up all the data from the left table(transactions) and join that with only any relavent data that has an intersection with the right table(customers) (transactions U customers')

     SELECT * FROM transactions RIGHT JOIN customers ON transactions.customer_id = customers.customer_id;
    //This will pull up all the data from the right table(customers) and join that with only any relavent data that has an intersection with the left table(transactions) (transactions' U customers)
    //In other words we will display all the data from the right table and if there are any matches pull up any rows from the left table
